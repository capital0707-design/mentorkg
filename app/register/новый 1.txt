'use client'
import { useState, useEffect } from 'react'
import { categories } from '@/lib/mock-data'
import Link from 'next/link'
import { useRouter } from 'next/navigation'

export default function Register() {
  const router = useRouter()
  const [category, setCategory] = useState('school')
  const [subject, setSubject] = useState('Математика')
  const [formData, setFormData] = useState({
    name: '', phone: '', level: '', price: 700,
    formats: ['online'] as string[], district: '',
    bio: '', education: ''
  })
  const [isSubmitted, setIsSubmitted] = useState(false)

  const currentSubjects = categories.find(c => c.id === category)?.subjects || []

  useEffect(() => {
    if (!currentSubjects.includes(subject)) setSubject(currentSubjects[0] || '')
  }, [category])

  const handleToggleFormat = (format: string) => {
    setFormData(prev => ({
      ...prev,
      formats: prev.formats.includes(format)
        ? prev.formats.filter(f => f !== format)
        : [...prev.formats, format]
    }))
  }

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault()
    
    // Сохраняем в localStorage (имитация базы)
    const newTutor = {
      id: Date.now(),
      photo_url: `https://i.pravatar.cc/150?u=${formData.name.replace(/\s/g, '').toLowerCase()}`,
      rating: 0, reviews_count: 0,
      category, subject,
      ...formData,
      reviews: []
    }
    
    const existing = JSON.parse(localStorage.getItem('tutors') || '[]')
    localStorage.setItem('tutors', JSON.stringify([...existing, newTutor]))
    
    setIsSubmitted(true)
    setTimeout(() => router.push('/'), 2000)
  }

  if (isSubmitted) {
    return (
      <main className="min-h-screen bg-gray-100 max-w-md mx-auto flex items-center justify-center p-4">
        <div className="bg-white rounded-2xl p-6 text-center shadow-lg border border-gray-200">
          <div className="text-5xl mb-4">✅</div>
          <h2 className="text-xl font-bold text-gray-900 mb-2">Заявка отправлена!</h2>
          <p className="text-gray-500 text-sm">Мы проверим данные и активируем профиль в течение 24 часов.</p>
        </div>
      </main>
    )
  }

  const inputClass = "w-full p-3 bg-white border border-gray-300 rounded-xl text-sm font-medium text-gray-900 focus:outline-none focus:ring-2 focus:ring-indigo-500 focus:border-indigo-500 transition"
  const labelClass = "block text-sm font-semibold text-gray-700 mb-2"

  return (
    <main className="min-h-screen bg-gray-100 max-w-md mx-auto pb-24">
      {/* Шапка */}
      <header className="sticky top-0 z-50 bg-white border-b border-gray-200 px-4 py-3 flex items-center gap-3">
        <Link href="/" className="p-2 -ml-2 rounded-full hover:bg-gray-100">←</Link>
        <h1 className="font-semibold text-gray-900">Регистрация репетитора</h1>
      </header>

      <form onSubmit={handleSubmit} className="p-4 space-y-4">
        {/* Контакты */}
        <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-200 space-y-4">
          <h2 className="font-bold text-gray-900 text-lg">👤 Контактные данные</h2>
          
          <div>
            <label className={labelClass}>Ваше имя *</label>
            <input required type="text" placeholder="Иван Иванов" 
              value={formData.name} onChange={e => setFormData({...formData, name: e.target.value})}
              className={inputClass} />
          </div>
          
          <div>
            <label className={labelClass}>Телефон *</label>
            <input required type="tel" placeholder="+996 700 123 456" 
              value={formData.phone} onChange={e => setFormData({...formData, phone: e.target.value})}
              className={inputClass} />
            <p className="text-xs text-gray-400 mt-1">На него придёт код подтверждения</p>
          </div>
        </div>

        {/* Предмет */}
        <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-200 space-y-4">
          <h2 className="font-bold text-gray-900 text-lg">📚 Что преподаёте</h2>
          
          <div>
            <label className={labelClass}>Категория *</label>
            <select value={category} onChange={e => setCategory(e.target.value)} className={inputClass}>
              {categories.map(c => (
                <option key={c.id} value={c.id}>{c.name}</option>
              ))}
            </select>
          </div>
          
          <div>
            <label className={labelClass}>Предмет *</label>
            <select value={subject} onChange={e => setSubject(e.target.value)} className={inputClass}>
              {currentSubjects.map(s => (
                <option key={s} value={s}>{s}</option>
              ))}
            </select>
          </div>
          
          <div>
            <label className={labelClass}>Уровень учеников *</label>
            <select value={formData.level} onChange={e => setFormData({...formData, level: e.target.value})} className={inputClass}>
              <option value="">Выберите уровень</option>
              <option value="Дошкольники (3-6 лет)">🧸 Дошкольники (3-6 лет)</option>
              <option value="Начальная школа (1-4 класс)">📓 Начальная школа (1-4 класс)</option>
              <option value="Средняя школа (5-9 класс)">📕 Средняя школа (5-9 класс)</option>
              <option value="Старшая школа (10-11 класс)">📗 Старшая школа (10-11 класс)</option>
              <option value="Студенты / ВУЗ">🎓 Студенты / ВУЗ</option>
              <option value="Взрослые">💼 Взрослые</option>
            </select>
          </div>
        </div>

        {/* Условия */}
        <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-200 space-y-4">
          <h2 className="font-bold text-gray-900 text-lg">⚙️ Условия занятий</h2>
          
          <div>
            <label className={labelClass}>Формат занятий *</label>
            <div className="grid grid-cols-3 gap-2">
              {[
                { id: 'online', icon: '💻', label: 'Онлайн' },
                { id: 'offline', icon: '🏠', label: 'Офлайн' },
                { id: 'micro', icon: '👥', label: 'Группа' }
              ].map(f => (
                <button 
                  key={f.id} type="button"
                  onClick={() => handleToggleFormat(f.id)}
                  className={`py-2.5 px-2 rounded-xl text-xs font-semibold transition-all border ${
                    formData.formats.includes(f.id)
                      ? 'bg-indigo-600 text-white border-indigo-600 shadow-lg' 
                      : 'bg-white text-gray-700 border-gray-300 hover:border-indigo-400'
                  }`}
                >
                  <span className="text-lg block">{f.icon}</span>
                  <span>{f.label}</span>
                </button>
              ))}
            </div>
          </div>
          
          <div>
            <label className={labelClass}>Район Бишкека (для офлайн)</label>
            <select value={formData.district} onChange={e => setFormData({...formData, district: e.target.value})} className={inputClass}>
              <option value="">Не важно / Только онлайн</option>
              <option value="Октябрьский">📍 Октябрьский</option>
              <option value="Свердловский">📍 Свердловский</option>
              <option value="Первомайский">📍 Первомайский</option>
              <option value="Ленинский">📍 Ленинский</option>
            </select>
          </div>
          
          <div>
            <label className={labelClass}>Цена за урок (сом) *</label>
            <select value={formData.price} onChange={e => setFormData({...formData, price: Number(e.target.value)})} className={inputClass}>
              <option value={400}>400 сом</option>
              <option value={600}>600 сом</option>
              <option value={700}>700 сом</option>
              <option value={800}>800 сом</option>
              <option value={900}>900 сом</option>
              <option value={1000}>1000 сом</option>
              <option value={1500}>1500 сом</option>
              <option value={2000}>2000 сом</option>
            </select>
          </div>
        </div>

        {/* О себе */}
        <div className="bg-white rounded-2xl p-4 shadow-sm border border-gray-200 space-y-4">
          <h2 className="font-bold text-gray-900 text-lg">📝 О себе</h2>
          
          <div>
            <label className={labelClass}>Образование</label>
            <input type="text" placeholder="КГУ, Физ-мат, 2019" 
              value={formData.education} onChange={e => setFormData({...formData, education: e.target.value})}
              className={inputClass} />
          </div>
          
          <div>
            <label className={labelClass}>Коротко об опыте *</label>
            <textarea required rows={3} placeholder="8 лет преподавания. Подготовка к ОРТ/НЦТ..." 
              value={formData.bio} onChange={e => setFormData({...formData, bio: e.target.value})}
              className={`${inputClass} resize-none`} />
          </div>
        </div>

        {/* Кнопка */}
        <button type="submit" 
          className="w-full bg-indigo-600 text-white py-4 rounded-xl font-bold text-base hover:bg-indigo-700 active:scale-[0.98] transition-all shadow-lg shadow-indigo-200">
          🚀 Отправить заявку
        </button>
        
        <p className="text-xs text-gray-400 text-center">
          Нажимая кнопку, вы соглашаетесь с обработкой персональных данных
        </p>
      </form>
    </main>
  )
}
